function transitionOpacity(element, newValue) { const wrapper = element.parentElement; // Получаем настройки из инпутов (по умолчанию 10% и 100 мс) const stepPercent = parseFloat(document.getElementById('stepPercent').value) || 10; const stepInterval = parseInt(document.getElementById('stepInterval').value) || 100; const stepRatio = stepPercent / 100; // Создаем элемент для набегающей цифры const newDigit = document.createElement('span'); newDigit.className = 'number'; newDigit.innerText = newValue; // Начальные состояния прозрачности: // Старая цифра уходит с 100% (1.0), новая появляется с 0% (0.0) element.style.opacity = '1'; newDigit.style.opacity = '0'; wrapper.appendChild(newDigit); let progress = 0; const interval = setInterval(() => { progress += stepRatio; if (progress > 1) progress = 1; // Старая цифра гаснет (1.0 -> 0.0) element.style.opacity = (1 - progress).toFixed(2); // Новая цифра проявляется (0.0 -> 1.0) newDigit.style.opacity = progress.toFixed(2); // Завершение анимации через 1000 мс (при 10 шагах по 100 мс) if (progress >= 1) { clearInterval(interval); element.innerText = newValue; element.style.opacity = '1'; newDigit.remove(); } }, stepInterval); }
