function processBrushStroke() { if (strokePath.length < 2) return; const canvas = document.getElementById('cv-canvas'); const ctx = canvas.getContext('2d'); const width = canvas.width; const height = canvas.height; if (cleanedImageSnapshot) { ctx.putImageData(cleanedImageSnapshot, 0, 0); } const imgData = ctx.getImageData(0, 0, width, height); const data = imgData.data; const pStart = strokePath[0]; const pEnd = strokePath[strokePath.length - 1]; const dx = Math.abs(pEnd.x - pStart.x); const dy = Math.abs(pEnd.y - pStart.y); let isHorizontal = dx > dy; // 1. Стираем старую кривую линию в зоне штриха strokePath.forEach(pt => { for (let rx = -10; rx <= 10; rx++) { for (let ry = -10; ry <= 10; ry++) { let px = pt.x + rx; let py = pt.y + ry; if (px >= 0 && px < width && py >= 0 && py < height) { let idx = (py * width + px) * 4; data[idx] = 255; data[idx+1] = 255; data[idx+2] = 255; } } } }); // 2. Строим ровную линию ТОЧНО по координате начала движения мыши (pStart), // чтобы она не смещалась в сторону. if (isHorizontal) { let fixedY = pStart.y; // Берем точную точку старта по Y let startX = Math.min(pStart.x, pEnd.x); let endX = Math.max(pStart.x, pEnd.x); for (let x = startX; x <= endX; x++) { for (let offY = -1; offY <= 1; offY++) { let py = fixedY + offY; if (py >= 0 && py < height) { let idx = (py * width + x) * 4; data[idx] = 0; data[idx+1] = 0; data[idx+2] = 0; } } } } else { let fixedX = pStart.x; // Берем точную точку старта по X let startY = Math.min(pStart.y, pEnd.y); let endY = Math.max(pStart.y, pEnd.y); for (let y = startY; y <= endY; y++) { for (let offX = -1; offX <= 1; offX++) { let px = fixedX + offX; if (px >= 0 && px < width) { let idx = (y * width + px) * 4; data[idx] = 0; data[idx+1] = 0; data[idx+2] = 0; } } } } ctx.putImageData(imgData, 0, 0); originalImageSnapshot = ctx.getImageData(0, 0, width, height); cleanedImageSnapshot = ctx.getImageData(0, 0, width, height); log(`✨ Линия выровнена точно по месту штриха (3px).`); strokePath = []; }
